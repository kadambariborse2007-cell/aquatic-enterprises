def require_admin(authorization: str = Header(None)):
    if not authorization or not authorization.startswith("Bearer "):
        raise HTTPException(status_code=401, detail="Admin login required.")

    token = authorization.split(" ", 1)[1]

    try:
        payload = decode_token(token)

        if payload.get("role") != "admin":
            raise HTTPException(status_code=403, detail="Admin access only.")

        email = payload.get("sub")

        with get_connection() as connection:
            admin = connection.execute(
                "SELECT id, name, email FROM admins WHERE email = %s",
                (email,)
            ).fetchone()

        if not admin:
            raise HTTPException(status_code=401, detail="Admin account not found.")

        return {
            "id": admin[0],
            "name": admin[1],
            "email": admin[2]
        }

    except HTTPException:
        raise
    except Exception:
        raise HTTPException(status_code=401, detail="Invalid or expired admin token.")


@router.post("/login")
def admin_login(data: AdminLoginData):
    email = str(data.email).strip().lower()

    with get_connection() as connection:
        admin = connection.execute(
            """
            SELECT id, name, email, password_hash
            FROM admins
            WHERE email = %s
            """,
            (email,)
        ).fetchone()

    if not admin or not verify_password(data.password, admin[3]):
        raise HTTPException(status_code=401, detail="Invalid admin email or password.")

    token = create_admin_token(admin[2])

    return {
        "success": True,
        "message": "Admin login successful.",
        "access_token": token,
        "token_type": "bearer",
        "admin": {
            "id": admin[0],
            "name": admin[1],
            "email": admin[2]
        }
    }

import os
import hmac
import hashlib
import secrets
import datetime

import jwt


def hash_password(password: str) -> str:
    salt = secrets.token_bytes(16)
    key = hashlib.scrypt(password.encode(), salt=salt, n=16384, r=8, p=1)
    return salt.hex() + ":" + key.hex()


def verify_password(password: str, stored: str) -> bool:
    salt_hex, key_hex = stored.split(":")
    key = hashlib.scrypt(
        password.encode(), salt=bytes.fromhex(salt_hex), n=16384, r=8, p=1
    )
    return hmac.compare_digest(key.hex(), key_hex)


def _secret() -> str:
    secret = os.getenv("JWT_SECRET")
    if not secret:
        raise RuntimeError("JWT_SECRET is not set.")
    return secret


def create_token(username: str) -> str:
    payload = {
        "sub": username,
        "exp": datetime.datetime.now(datetime.timezone.utc)
        + datetime.timedelta(hours=8),
    }
    return jwt.encode(payload, _secret(), algorithm="HS256")


def decode_token(token: str) -> dict:
    return jwt.decode(token, _secret(), algorithms=["HS256"])

def create_admin_token(email: str) -> str:
    payload = {
        "sub": email,
        "role": "admin",
        "exp": datetime.datetime.now(datetime.timezone.utc)
        + datetime.timedelta(hours=8),
    }
    return jwt.encode(payload, _secret(), algorithm="HS256")

@app.post("/register")
def register(data: RegisterData):

    email = str(data.email).strip().lower()

    if len(data.password) < 8:
        raise HTTPException(
            status_code=400,
            detail="Password must be at least 8 characters."
        )

    try:
        with get_connection() as connection:

            exists = connection.execute(
                "SELECT id FROM customers WHERE email = %s",
                (email,)
            ).fetchone()

            if exists:
                raise HTTPException(
                    status_code=400,
                    detail="Email already registered."
                )

            connection.execute(
                """
                INSERT INTO customers
                (name, email, phone, password_hash)
                VALUES (%s, %s, %s, %s)
                """,
                (
                    data.name.strip(),
                    email,
                    data.phone,
                    hash_password(data.password)
                )
            )

            connection.commit()

        return {
            "token": create_token(email),
            "name": data.name.strip()
        }

    except HTTPException:
        raise

    except Exception as error:

        print("Registration error:", repr(error))

        raise HTTPException(
            status_code=500,
            detail="Unable to register customer."

"""
Customer reviews API (FastAPI + PostgreSQL).

Flow: visitor submits a review -> saved as "pending" -> admin approves it ->
only "approved" reviews appear on the website.

Uses the same get_connection() as the rest of the project.
Needs ADMIN_KEY in backend/.env (temporary admin protection).
"""

import hmac
import os
import time
from collections import defaultdict, deque
from typing import Literal, Optional

from fastapi import APIRouter, Depends, Header, HTTPException, Request
from pydantic import BaseModel, ConfigDict, Field

from database import get_connection

try:
    from dotenv import load_dotenv

    load_dotenv()
except ImportError:
    pass

router = APIRouter()

# ---------- table (created automatically on first use) ----------
_table_ready = False


def _ensure_table(connection) -> None:
    global _table_ready
    if _table_ready:
        return
    connection.execute(
        """
        CREATE TABLE IF NOT EXISTS reviews (
            id         SERIAL PRIMARY KEY,
            name       VARCHAR(80)  NOT NULL,
            rating     SMALLINT     NOT NULL CHECK (rating BETWEEN 1 AND 5),
            comment    TEXT         NOT NULL,
            status     VARCHAR(10)  NOT NULL DEFAULT 'pending',
            created_at TIMESTAMPTZ  NOT NULL DEFAULT NOW()
        )
        """
    )
    connection.commit()
    _table_ready = True


# ---------- schemas ----------
class ReviewIn(BaseModel):
    model_config = ConfigDict(str_strip_whitespace=True)

    name: str = Field(min_length=2, max_length=80)
    rating: int = Field(ge=1, le=5)
    comment: str = Field(min_length=5, max_length=1000)


class StatusIn(BaseModel):
    status: Literal["pending", "approved", "rejected"]


# ---------- spam protection: 3 reviews per IP per hour ----------
RATE_LIMIT = 3
RATE_WINDOW = 3600
_recent = defaultdict(deque)


def _check_rate_limit(ip: str) -> None:
    now = time.time()
    queue = _recent[ip]
    while queue and now - queue[0] > RATE_WINDOW:
        queue.popleft()
    if len(queue) >= RATE_LIMIT:
        raise HTTPException(429, "Too many reviews. Please try again later.")
    queue.append(now)


# ---------- temporary admin protection ----------
def require_admin_key(x_admin_key: Optional[str] = Header(default=None)) -> None:
    expected = os.getenv("ADMIN_KEY")
    if not expected or not x_admin_key or not hmac.compare_digest(x_admin_key, expected):
        raise HTTPException(401, "Admin access required.")


def _to_dict(row, with_status: bool = False) -> dict:
    data = {
        "id": row[0],
        "name": row[1],
        "rating": row[2],
        "comment": row[3],
    }
    if with_status:
        data["status"] = row[4]
        data["created_at"] = row[5].isoformat()
    else:
        data["created_at"] = row[4].isoformat()
    return data


def _server_error(label: str, error: Exception):
    print(f"{label}:", repr(error), flush=True)
    return HTTPException(500, "Something went wrong. Please try again.")


# ---------- public endpoints ----------
@router.post("/reviews", status_code=201)
def create_review(review: ReviewIn, request: Request):
    ip = request.client.host if request.client else "unknown"
    _check_rate_limit(ip)

    try:
        with get_connection() as connection:
            _ensure_table(connection)
            connection.execute(
                "INSERT INTO reviews (name, rating, comment) VALUES (%s, %s, %s)",
                (review.name, review.rating, review.comment),
            )
            connection.commit()
    except Exception as error:
        raise _server_error("Review submit error", error)

    return {"message": "Thank you! Your review will appear after approval."}


@router.get("/reviews")
def list_approved_reviews(limit: int = 50):
    limit = max(1, min(limit, 100))
    try:
        with get_connection() as connection:
            _ensure_table(connection)
            rows = connection.execute(
                "SELECT id, name, rating, comment, created_at FROM reviews "
                "WHERE status = 'approved' ORDER BY created_at DESC LIMIT %s",
                (limit,),
            ).fetchall()
    except Exception as error:
        raise _server_error("Review list error", error)

    return [_to_dict(row) for row in rows]


@router.get("/reviews/summary")
def reviews_summary():
    try:
        with get_connection() as connection:
            _ensure_table(connection)
            rows = connection.execute(
                "SELECT rating, COUNT(*) FROM reviews "
                "WHERE status = 'approved' GROUP BY rating"
            ).fetchall()
    except Exception as error:
        raise _server_error("Review summary error", error)

    breakdown = {str(star): 0 for star in range(1, 6)}
    for rating, total in rows:
        breakdown[str(rating)] = total

    count = sum(breakdown.values())
    average = (
        round(sum(int(star) * total for star, total in breakdown.items()) / count, 1)
        if count
        else 0
    )
    return {"average": average, "count": count, "breakdown": breakdown}


# ---------- admin endpoints ----------
@router.get("/admin/reviews", dependencies=[Depends(require_admin_key)])
def admin_list_reviews(status: Optional[str] = None):
    query = "SELECT id, name, rating, comment, status, created_at FROM reviews"
    params = ()
    if status:
        query += " WHERE status = %s"
        params = (status,)
    query += " ORDER BY created_at DESC"

    try:
        with get_connection() as connection:
            _ensure_table(connection)
            rows = connection.execute(query, params).fetchall()
    except Exception as error:
        raise _server_error("Admin review list error", error)

    return [_to_dict(row, with_status=True) for row in rows]


@router.patch("/admin/reviews/{review_id}", dependencies=[Depends(require_admin_key)])
def admin_set_status(review_id: int, body: StatusIn):
    try:
        with get_connection() as connection:
            _ensure_table(connection)
            result = connection.execute(
                "UPDATE reviews SET status = %s WHERE id = %s",
                (body.status, review_id),
            )
            updated = result.rowcount
            connection.commit()
    except Exception as error:
        raise _server_error("Admin review update error", error)

    if updated == 0:
        raise HTTPException(404, "Review not found.")
    return {"id": review_id, "status": body.status}


@router.delete("/admin/reviews/{review_id}", dependencies=[Depends(require_admin_key)])
def admin_delete_review(review_id: int):
    try:
        with get_connection() as connection:
            _ensure_table(connection)
            result = connection.execute("DELETE FROM reviews WHERE id = %s", (review_id,))
            deleted = result.rowcount
            connection.commit()
    except Exception as error:
        raise _server_error("Admin review delete error", error)

    if deleted == 0:
        raise HTTPException(404, "Review not found.")
    return {"deleted": review_id}

import { createContext, useContext, useState, useEffect } from "react";

const API = "http://127.0.0.1:8001";
const AuthContext = createContext(null);

async function post(path, body) {
  const res = await fetch(`${API}${path}`, {
    method: "POST",
    headers: { "Content-Type": "application/json" },
    body: JSON.stringify(body),
  });
  const data = await res.json().catch(() => ({}));
  if (!res.ok) {
    const d = data.detail;
    throw new Error(typeof d === "string" ? d : "Please check your details and try again.");
  }
  return data;
}

export function AuthProvider({ children }) {
  const [token, setToken] = useState(localStorage.getItem("token"));
  const [user, setUser] = useState(null);
  const [loading, setLoading] = useState(!!localStorage.getItem("token"));

  useEffect(() => {
    if (!token) {
      setUser(null);
      setLoading(false);
      return;
    }
    fetch(`${API}/me`, { headers: { Authorization: `Bearer ${token}` } })
      .then((r) => {
        if (!r.ok) throw new Error();
        return r.json();
      })
      .then(setUser)
      .catch(() => {
        localStorage.removeItem("token");
        setToken(null);
        setUser(null);
      })
      .finally(() => setLoading(false));
  }, [token]);

  const saveToken = (t) => {
    localStorage.setItem("token", t);
    setToken(t);
  };

  const login = async (email, password) => {
    const data = await post("/login", { email, password });
    saveToken(data.token);
  };

  const register = async (name, email, phone, password) => {
    const data = await post("/register", { name, email, phone, password });
    saveToken(data.token);
  };

  const logout = () => {
    localStorage.removeItem("token");
    setToken(null);
    setUser(null);
  };

  return (
    <AuthContext.Provider value={{ user, token, loading, login, register, logout }}>
      {children}
    </AuthContext.Provider>
  );
}

export function useAuth() {
  return useContext(AuthContext);
}

export default function AdminLogin() {
  const navigate = useNavigate();

    setError("");

      navigate("/admin/dashboard");
    } catch (error) {
      setError(error.message || "Unable to connect to the server.");
    } finally {
      setLoading(false);
    }
  }

  return (
    <div style={styles.page}>
      <form onSubmit={handleLogin} style={styles.card}>
        <h1 style={styles.heading}>Aquatic Enterprises</h1>
        <p style={styles.subtitle}>Administrator Login</p>

        {error && <p style={styles.error}>{error}</p>}

        <label style={styles.label}>Admin Email</label>
        <input
          type="email"
          value={email}
          onChange={(event) => setEmail(event.target.value)}
          placeholder="Enter admin email"
          required
          style={styles.input}
        />

        <label style={styles.label}>Password</label>
        <input
          type="password"
          value={password}
          onChange={(event) => setPassword(event.target.value)}
          placeholder="Enter admin password"
          required
          style={styles.input}
        />

        <button type="submit" disabled={loading} style={styles.button}>
          {loading ? "Logging in..." : "Login to Dashboard"}
        </button>
      </form>
    </div>
  );
}
